project(CHECKPOINT_ROUNDTRIP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
list(PREPEND CMAKE_PREFIX_PATH /home/lxq/miniconda3/envs/cuvs_benchmarks)
